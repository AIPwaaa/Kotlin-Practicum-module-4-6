class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            if (checkSelfPermission(android.Manifest.permission.POST_NOTIFICATIONS)
                != PackageManager.PERMISSION_GRANTED) {

                requestPermissions(
                    arrayOf(android.Manifest.permission.POST_NOTIFICATIONS),
                    100
                )
            }
        }

        NotificationHelper.createChannel(this)

        setContent {
            Screen { startWeatherChain() }
        }
    }

    private fun startWeatherChain() {
        val moscowRequest = OneTimeWorkRequestBuilder<WeatherWorker>()
            .setInputData(workDataOf("city_name" to "Москва"))
            .build()

        val londonRequest = OneTimeWorkRequestBuilder<WeatherWorker>()
            .setInputData(workDataOf("city_name" to "Лондон"))
            .build()

        val newYorkRequest = OneTimeWorkRequestBuilder<WeatherWorker>()
            .setInputData(workDataOf("city_name" to "Нью-Йорк"))
            .build()

        val reportRequest = OneTimeWorkRequestBuilder<ReportWorker>().build()

        WorkManager.getInstance(this)
            .beginWith(listOf(moscowRequest, londonRequest, newYorkRequest))
            .then(reportRequest)
            .enqueue()
    }
}

@Composable
fun Screen(onStartClick: () -> Unit) {
    Surface(
        modifier = Modifier.fillMaxSize(),
        color = MaterialTheme.colorScheme.background
    ) {
        Column(
            modifier = Modifier.fillMaxSize(),
            verticalArrangement = Arrangement.Center,
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Text(
                text = "Сбор прогноза погоды",
                style = MaterialTheme.typography.headlineMedium
            )
            Spacer(modifier = Modifier.height(20.dp))
            Button(onClick = onStartClick) {
                Text("Собрать прогноз")
            }
        }
    }
}

package com.example.myapplication

import android.app.NotificationChannel
import android.app.NotificationManager
import androidx.work.Worker
import androidx.work.WorkerParameters
import android.content.Context
import android.os.Build
import androidx.core.app.NotificationCompat
import androidx.work.CoroutineWorker
import androidx.work.ForegroundInfo
import androidx.work.workDataOf
import kotlinx.coroutines.delay

class WeatherWorker(context: Context, params: WorkerParameters) : CoroutineWorker(context, params) {
    suspend override fun doWork(): Result {
        val cityName = inputData.getString("city_name") ?: "Unknown"

        setForeground(createForegroundInfo("Загрузка: $cityName..."))

        delay((2000..5000).random().toLong())
        val temp = (10..25).random()

        val outputData = workDataOf(cityName to temp)
        return Result.success(outputData)
    }
    private fun createForegroundInfo(text: String): ForegroundInfo {
        val notification = NotificationCompat.Builder(applicationContext, NotificationHelper.CHANNEL_ID)
            .setContentTitle("Сбор данных")
            .setContentText(text)
            .setSmallIcon(android.R.drawable.ic_menu_agenda)
            .setOngoing(true)
            .build()

        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            ForegroundInfo(
                NotificationHelper.NOTIFICATION_ID,
                notification,
                android.content.pm.ServiceInfo.FOREGROUND_SERVICE_TYPE_DATA_SYNC
            )
        } else {
            ForegroundInfo(NotificationHelper.NOTIFICATION_ID, notification)
        }
    }
}

package com.example.myapplication

import android.app.NotificationManager
import android.content.Context
import android.os.Build
import androidx.core.app.NotificationCompat
import androidx.work.CoroutineWorker
import androidx.work.ForegroundInfo
import androidx.work.WorkerParameters

class ReportWorker(context: Context, params: WorkerParameters) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result {
        setForeground(createForegroundInfo("Все данные получены, формируем отчёт…"))

        kotlinx.coroutines.delay(2000)

        val allData = inputData.keyValueMap
        val temperatures = allData.values.filterIsInstance<Int>()

        val avgTemp = if (temperatures.isNotEmpty()) temperatures.average().toInt() else 0
        val finalResult = "Отчёт готов! Средняя температура: +$avgTemp°C"

        val notificationManager = applicationContext.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        val finalNotification = NotificationCompat.Builder(applicationContext, NotificationHelper.CHANNEL_ID)
            .setContentTitle("Погода собрана")
            .setContentText(finalResult)
            .setSmallIcon(android.R.drawable.ic_menu_compass)
            .setPriority(NotificationCompat.PRIORITY_DEFAULT)
            .setOngoing(false)
            .build()

        notificationManager.notify(NotificationHelper.NOTIFICATION_ID, finalNotification)

        return Result.success()
    }

    private fun createForegroundInfo(text: String): ForegroundInfo {
        val notification = NotificationCompat.Builder(applicationContext, NotificationHelper.CHANNEL_ID)
            .setContentTitle("Сбор данных")
            .setContentText(text)
            .setSmallIcon(android.R.drawable.ic_menu_agenda)
            .setOngoing(true)
            .build()

        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            ForegroundInfo(
                NotificationHelper.NOTIFICATION_ID,
                notification,
                android.content.pm.ServiceInfo.FOREGROUND_SERVICE_TYPE_DATA_SYNC
            )
        } else {
            ForegroundInfo(NotificationHelper.NOTIFICATION_ID, notification)
        }
    }
}

package com.example.myapplication

import android.app.NotificationChannel
import android.app.NotificationManager
import android.content.Context
import android.os.Build

object NotificationHelper {
    const val CHANNEL_ID = "weather_channel"
    const val NOTIFICATION_ID = 101

    fun createChannel(context: Context) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val name = "Weather Updates"
            val descriptionText = "Уведомления о загрузке погоды"
            val importance = NotificationManager.IMPORTANCE_LOW
            val channel = NotificationChannel(CHANNEL_ID, name, importance).apply {
                description = descriptionText
            }
            val notificationManager = context.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            notificationManager.createNotificationChannel(channel)
        }
    }
}
