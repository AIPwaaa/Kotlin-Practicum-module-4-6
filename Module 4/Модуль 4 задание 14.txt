class MainActivity : ComponentActivity() {

    private lateinit var sensorManager: SensorManager
    private var accelerometer: Sensor? = null
    private var magnetometer: Sensor? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        sensorManager = getSystemService(Context.SENSOR_SERVICE) as SensorManager
        accelerometer = sensorManager.getDefaultSensor(Sensor.TYPE_ACCELEROMETER)
        magnetometer = sensorManager.getDefaultSensor(Sensor.TYPE_MAGNETIC_FIELD)

        if(accelerometer == null || magnetometer == null) {
            Toast.makeText(this, "Датчики не найдены!", Toast.LENGTH_LONG).show()
        }

        setContent {
            Screen()
        }
    }

    override fun onResume() {
        super.onResume()

        val vm: CompassViewModel by viewModels()
        accelerometer?.let {
            sensorManager.registerListener(vm, it, SensorManager.SENSOR_DELAY_UI)
        }
        magnetometer?.let {
            sensorManager.registerListener(vm, it, SensorManager.SENSOR_DELAY_UI)
        }
    }

    override fun onPause() {
        super.onPause()
        val vm: CompassViewModel by viewModels()
        sensorManager.unregisterListener(vm)
    }
}

@Composable
fun Screen(vm: CompassViewModel = viewModel()) {
    val azimuth by vm.azimuth.collectAsState()

    val animatedAzimuth by animateFloatAsState(
        targetValue = azimuth,
        animationSpec = spring(stiffness = Spring.StiffnessLow),
        label = "compassRotation"
    )

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color(0xFF121212)),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center
    ) {
        Text(
            text = "${azimuth.toInt().coerceIn(0, 360)}°",
            color = Color.White,
            fontSize = 48.sp,
            style = MaterialTheme.typography.displayLarge
        )

        Spacer(modifier = Modifier.height(32.dp))

        Box(
            modifier = Modifier
                .fillMaxWidth(0.8f)
                .aspectRatio(1f)
                .drawBehind {
                    drawCircle(
                        color = Color.Gray,
                        style = Stroke(width = 8f)
                    )
                },
            contentAlignment = Alignment.Center
        ) {
            Text(
                text = "N",
                color = Color.Red,
                fontSize = 24.sp,
                modifier = Modifier
                    .align(Alignment.TopCenter)
                    .padding(top = 10.dp)
            )

            Canvas(modifier = Modifier
                .fillMaxSize(0.7f)
                .rotate(animatedAzimuth)
            ) {
                val center = size / 2f
                val needleWidth = 15f

                val northPath = Path().apply {
                    moveTo(center.width, 0f)
                    lineTo(center.width - needleWidth, center.height)
                    lineTo(center.width + needleWidth, center.height)
                    close()
                }
                drawPath(northPath, Color.Red)

                val southPath = Path().apply {
                    moveTo(center.width, size.height)
                    lineTo(center.width - needleWidth, center.height)
                    lineTo(center.width + needleWidth, center.height)
                    close()
                }
                drawPath(southPath, Color.Gray)
            }
        }
    }
}
